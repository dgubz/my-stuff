Iframe Detector POC

Click anywhere to detect if this page is inside an iframe. After clicking, an alert will indicate whether this page is inside an iframe or not.

This could work better than traditional iframe detection methods because it does not rely on JavaScript properties that can be easily spoofed or blocked by the browser.

It utilizes the favicon request to determine if the page is inside an. iframe, so your page requires a favicon to be specified in the <head> section.

For this demo, check console for the detection result.


To use this on your page, use the JavaScript in the code block below


const faviconLink = document.querySelector("link[rel='shortcut icon'], link[rel='icon']");
const faviconUrl = new URL(faviconLink.href, window.location.href).href;
let faviconSeen = false;

const isFaviconRequest = (resourceUrl) => {
  const resolvedResourceUrl = new URL(resourceUrl, window.location.href).href;
  return resolvedResourceUrl === faviconUrl;
};

if (PerformanceObserver) {
  const observer = new PerformanceObserver((list) => {
    for (const entry of list.getEntries()) {
      if (isFaviconRequest(entry.name)) {
        faviconSeen = true;
        observer.disconnect();
        isNotInIframe();
        return;
      }
    }
  });

  observer.observe({ type: 'resource', buffered: true });
}

window.addEventListener('load', (event) => {
  const delay = performance.now();
  const wait = setTimeout(() => {
    if (!faviconSeen) {
      isInIframe();
    }
  }, delay);
});

function isInIframe() {
  // perform decision
  console.log('This page is inside an iframe.');
}

function isNotInIframe() {
  // perform decision
  console.log('This page is not inside an iframe.');
}
